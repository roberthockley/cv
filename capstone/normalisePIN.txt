const raw = String(input.text || "").toLowerCase().trim();
api.log("debug", "Raw: " + raw);

const digitWords = {
  null: "0",
  oh: "0",
  ein: "1",
  eins: "1",
  eine: "1",
  zwei: "2",
  drei: "3",
  vier: "4",
  fünf: "5",
  fuenf: "5",
  sechs: "6",
  sieben: "7",
  acht: "8",
  neun: "9"
};

const teenTensWords = {
  zehn: "10",
  elf: "11",
  zwölf: "12",
  zwoelf: "12",
  dreizehn: "13",
  vierzehn: "14",
  fünfzehn: "15",
  fuenfzehn: "15",
  sechzehn: "16",
  siebzehn: "17",
  achtzehn: "18",
  neunzehn: "19",
  zwanzig: "20",
  dreißig: "30",
  dreissig: "30",
  vierzig: "40",
  fünfzig: "50",
  fuenfzig: "50",
  sechzig: "60",
  siebzig: "70",
  achtzig: "80",
  neunzig: "90"
};

const tensValue = {
  zwanzig: 20,
  dreißig: 30,
  dreissig: 30,
  vierzig: 40,
  fünfzig: 50,
  fuenfzig: 50,
  sechzig: 60,
  siebzig: 70,
  achtzig: 80,
  neunzig: 90
};

const multipliers = {
  doppelt: 2,
  zweifach: 2,
  dreifach: 3,
  vierfach: 4
};

function normalizeWord(word) {
  return String(word || "")
    .replace(/ä/g, "ae")
    .replace(/ö/g, "oe")
    .replace(/ü/g, "ue")
    .replace(/ß/g, "ss");
}

const cleaned = raw
  .replace(/[-–—]/g, " ")
  .replace(/\b(bindestrich|strich)\b/g, " ")
  .replace(/[^\w\säöüß]/g, " ")
  .replace(/\s+/g, " ")
  .trim();

const parts = cleaned.split(" ").filter(Boolean);
api.log("debug", "parts: " + parts);

context.security = context.security || {};

function getDigit(word) {
  if (!word) return undefined;
  if (digitWords[word] !== undefined) return digitWords[word];
  const normalized = normalizeWord(word);
  if (digitWords[normalized] !== undefined) return digitWords[normalized];
  return undefined;
}

function getTeenTens(word) {
  if (!word) return undefined;
  if (teenTensWords[word] !== undefined) return teenTensWords[word];
  const normalized = normalizeWord(word);
  if (teenTensWords[normalized] !== undefined) return teenTensWords[normalized];
  return undefined;
}

function getTensValue(word) {
  if (!word) return undefined;
  if (tensValue[word] !== undefined) return tensValue[word];
  const normalized = normalizeWord(word);
  if (tensValue[normalized] !== undefined) return tensValue[normalized];
  return undefined;
}

function getMultiplier(word) {
  if (!word) return undefined;
  if (multipliers[word] !== undefined) return multipliers[word];
  const normalized = normalizeWord(word);
  if (multipliers[normalized] !== undefined) return multipliers[normalized];
  return undefined;
}

function getOptions(tokens, i) {
  const out = [];
  const part = tokens[i];
  const next = tokens[i + 1];
  const next2 = tokens[i + 2];
  const next3 = tokens[i + 3];

  if (!part) return out;

  // 1) numeric chunk like "482913"
  if (/^\d+$/.test(part)) {
    out.push({ value: part, consumed: 1 });
  }

  // 2) multiplier like "doppelt fünf", "dreifach 2"
  const multiplier = getMultiplier(part);
  if (multiplier !== undefined && next !== undefined) {
    if (/^\d$/.test(next)) {
      out.push({ value: next.repeat(multiplier), consumed: 2 });
    }

    const nextDigit = getDigit(next);
    if (nextDigit !== undefined) {
      out.push({ value: nextDigit.repeat(multiplier), consumed: 2 });
    }
  }

  // 3) digit word like "vier"
  const partDigit = getDigit(part);
  if (partDigit !== undefined) {
    out.push({ value: partDigit, consumed: 1 });
  }

  // 4) teen/tens alone like "dreizehn", "achtzig"
  const partTeenTens = getTeenTens(part);
  if (partTeenTens !== undefined) {
    out.push({ value: partTeenTens, consumed: 1 });
  }

  // 5) tens + digit like "achtzig zwei" -> 82
  const partTensValue = getTensValue(part);
  const nextDigit = getDigit(next);
  if (
    partTensValue !== undefined &&
    nextDigit !== undefined &&
    nextDigit !== "0"
  ) {
    out.push({
      value: String(partTensValue + Number(nextDigit)),
      consumed: 2
    });
  }

  // 6) "vier hundert" -> 400
  if (
    partDigit !== undefined &&
    partDigit !== "0" &&
    next === "hundert"
  ) {
    out.push({
      value: String(Number(partDigit) * 100),
      consumed: 2
    });
  }

  // 7) "vier hundert achtzig" -> 480
  //    "vier hundert achtzig zwei" -> 482
  const next2TensValue = getTensValue(next2);
  if (
    partDigit !== undefined &&
    partDigit !== "0" &&
    next === "hundert" &&
    next2TensValue !== undefined
  ) {
    out.push({
      value: String(Number(partDigit) * 100 + next2TensValue),
      consumed: 3
    });

    const next3Digit = getDigit(next3);
    if (next3Digit !== undefined && next3Digit !== "0") {
      out.push({
        value: String(Number(partDigit) * 100 + next2TensValue + Number(next3Digit)),
        consumed: 4
      });
    }
  }

  // 8) "neun hundert dreizehn" -> 913
  const next2TeenTens = getTeenTens(next2);
  if (
    partDigit !== undefined &&
    partDigit !== "0" &&
    next === "hundert" &&
    next2TeenTens !== undefined
  ) {
    out.push({
      value: String(Number(partDigit) * 100 + Number(next2TeenTens)),
      consumed: 3
    });
  }

  return out;
}

function solve(tokens, index, built, memo) {
  const key = index + "|" + built;
  if (memo[key] !== undefined) return memo[key];

  if (built.length > 6) {
    memo[key] = null;
    return null;
  }

  if (index >= tokens.length) {
    memo[key] = built.length === 6 ? built : null;
    return memo[key];
  }

  const options = getOptions(tokens, index);

  // Prefer richer parses first
  options.sort((a, b) => {
    if (b.value.length !== a.value.length) return b.value.length - a.value.length;
    return b.consumed - a.consumed;
  });

  for (const option of options) {
    const result = solve(tokens, index + option.consumed, built + option.value, memo);
    if (result) {
      memo[key] = result;
      return result;
    }
  }

  // Ignore filler token
  const skipResult = solve(tokens, index + 1, built, memo);
  memo[key] = skipResult;
  return skipResult;
}

const digits = solve(parts, 0, "", {}) || "";

if (/^\d{6}$/.test(digits)) {
  context.security.pinCandidate = digits;
  context.security.pinDetected = true;
  context.security.pinNormalized = true;
} else {
  context.security.pinCandidate = null;
  context.security.pinDetected = false;
  context.security.pinNormalized = false;
}

api.log("debug", "Digits: " + digits);
api.log("debug", "Normalized PIN: " + (context.security.pinCandidate || "N/A"));
